# ASCII only, on purpose. pip reads this file as UTF-8, and a single stray
# byte from a Windows-1252 editor makes the whole install fail with a decode
# error that says nothing about which line is at fault. A requirements file
# has no need for a character outside ASCII, so it does not get one.
PyYAML>=6.0.3
# The floor is the version pylock.toml proves: Docker and CI install the lock,
# so a newer floor here means Windows resolves a client the tested stack never
# ran. Raise it only together with a regenerated lock.
qdrant-client>=1.19.1
# --- DEV ONLY below (test suite + coverage gate). Documented dev install is
# `pip install -e .[dev]` (see pyproject.toml); these rows stay until
# pylock.toml is regenerated on Linux, because Docker and CI install from
# that lock and this file is its human-readable source.
pytest>=9.1.1
# Coverage is a gate input, not a nicety: SonarCloud's quality gate requires
# 80% on new code, and with no report uploaded it reads 0% and fails.
pytest-cov>=7.1.0

# tested stack never ran. scripts/check_lock_parity.py fails if this drifts.
tzdata>=2026.4

# Live Ollama backend (s37.16) - the installed SDK returns pydantic responses.
